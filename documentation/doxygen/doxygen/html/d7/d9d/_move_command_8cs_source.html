120 private static string GetMoveDescription(
121 IImmutableDictionary<uint, Length>? linearPositions,
122 Speed? linearSpeed,
123 IImmutableDictionary<uint, Angle>? rotationalPositions,
124 RotationalSpeed? rotationalSpeed)
125 {
126 string speedString;
127 if (linearSpeed.HasValue)
128 speedString = "at " + linearSpeed.Value.ToString();
129 else if (rotationalSpeed.HasValue)
130 speedString = "at " + rotationalSpeed.Value.ToString();
131 else
132 speedString = "at current speed.";
133
134 bool hasLinearPositions = linearPositions is not null && linearPositions.Any();
135 bool hasRotationalPositions = rotationalPositions is not null && rotationalPositions.Any();
136 string linearPositionsString = $"linear positions: {(linearPositions is not null ? GetPositionsString(linearPositions) : string.Empty)}";
137 string rotationalPositionsString = $"rotational positions: {(rotationalPositions is not null ? GetPositionsString(rotationalPositions) : string.Empty)}";
138 if (hasLinearPositions && hasRotationalPositions)
139 return $"Move to {linearPositionsString} {rotationalPositionsString} {speedString}";
140 else if (hasLinearPositions)
141 return $"Move to {linearPositionsString} {speedString}";
142 else if (hasRotationalPositions)
143 return $"Move to {rotationalPositionsString} {speedString}";
144 else
145 return $"Move missing positions";
146 }

154 public Result<Move> GetMove(Machine initialMachine)
155 {
156 CompositeKinematicSystem kinematicSystem = initialMachine.MotionSystem.CompositeKinematicSystem;
157
158 // Get the final positions of all axes involved in the move, using the current position if no new position is specified.
159 List<Length?> positionsFinalLinear = [];
160 foreach (AxisLinear axis in kinematicSystem.AxesLinear)
161 {
162 if (PositionsLinear.TryGetValue(axis.Id, out Length pos))
163 positionsFinalLinear.Add(pos);
164 else
165 positionsFinalLinear.Add(axis.Pos);
166 }
167
168 // Repeat with rotational axes.
169 List<Angle?> positionsFinalRotational = [];
170 foreach (AxisRotational axis in kinematicSystem.AxesRotational)
171 {
172 if (PositionsRotational.TryGetValue(axis.Id, out Angle pos))
173 positionsFinalRotational.Add(pos);
174 else
175 positionsFinalRotational.Add(axis.Pos);
176 }
177
178 return Move.New(initialMachine.MotionSystem.Precisions,
179 kinematicSystem,
180 [.. kinematicSystem.AxesLinear.Select(axis => axis.Pos)],
181 [.. kinematicSystem.AxesRotational.Select(axis => axis.Pos)],
182 positionsFinalLinear,
183 positionsFinalRotational,
184 RequestedSpeedLinear,
185 RequestedSpeedRotational);
186 }
