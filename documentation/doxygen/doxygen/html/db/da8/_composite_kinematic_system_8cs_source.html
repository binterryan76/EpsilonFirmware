98 public Result<(IImmutableList<Length?>, IImmutableList<Angle?>)> ForwardKinematics(
99 IReadOnlyList<Length?> actuatorPositionsLinear,
100 IReadOnlyList<Angle?> actuatorPositionsRotational)
101 {
102 if (actuatorPositionsLinear.Count != ActuatorPositionsLinear.Length)
103 return new ArgumentException($"{ActuatorPositionsLinear.Length} linear actuators are required but {actuatorPositionsLinear.Count} were provided.");
104
105 if (actuatorPositionsRotational.Count != ActuatorPositionsRotational.Length)
106 return new ArgumentException($"{ActuatorPositionsRotational.Length} rotational actuators are required but {actuatorPositionsRotational.Count} were provided.");
107
108 // enter input data into private input arrays since the memory objects are already pointing to these lists
109 for (int i = 0; i < actuatorPositionsLinear.Count; i++)
110 ActuatorPositionsLinear[i] = actuatorPositionsLinear[i];
111
112 for (int i = 0; i < actuatorPositionsRotational.Count; i++)
113 ActuatorPositionsRotational[i] = actuatorPositionsRotational[i];
114
115 // run inverse kinematics on everything, any kinematics with a null input will be skipped and will output null
116 for (int i = 0; i < Kinematics.Count; i++)
117 {
118 Exception? error = Kinematics[i].ForwardKinematics(
119 ActuatorMemoriesLinear[i].Span,
120 ActuatorMemoriesRotational[i].Span,
121 AxisMemoriesLinear[i],
122 AxisMemoriesRotational[i]);
123
124 if (error is not null)
125 return error;
126 }
127
128 return (AxisPositionsLinear.ToImmutableList(), AxisPositionsRotational.ToImmutableList());
129 }

184 internal Result<CompositeKinematicSystem> AddKinematics(
185 Machine initialMachine,
186 IKinematics kinematicsToAdd,
187 IReadOnlyList<uint> actuatorIdsToAddLinear,
188 IReadOnlyList<uint> actuatorIdsToAddRotational,
189 IReadOnlyList<uint> axisIdsToAddLinear,
190 IReadOnlyList<uint> axisIdsToAddRotational,
191 IReadOnlyList<EndstopSet> endstopSetsToAdd)
192 {
193 // First validate inputs.
194 // Make sure the number of actuators and axes matches the degrees of freedom.
195 int actuatorsToAddCount = actuatorIdsToAddLinear.Count + actuatorIdsToAddRotational.Count;
196 int axesToAddCount = axisIdsToAddLinear.Count + axisIdsToAddRotational.Count;
197
198 if (kinematicsToAdd.DegreesOfFreedom != actuatorsToAddCount)
199 return new ArgumentException(
200 $"Cannot add kinematics '{kinematicsToAdd}' because it's degrees of freedom ({kinematicsToAdd.DegreesOfFreedom}) must match the number of actuators ({actuatorsToAddCount}).");
201
202 if (kinematicsToAdd.DegreesOfFreedom != axesToAddCount)
203 return new ArgumentException(
204 $"Cannot add kinematics '{kinematicsToAdd}' because it's degrees of freedom ({kinematicsToAdd.DegreesOfFreedom}) must match the number of axes ({axesToAddCount}).");
205
206 // Ensure axes and actuators have been added to the machine.
207 IEnumerable<uint> missingActuatorIdsLinear = actuatorIdsToAddLinear.Except(initialMachine.Entities.ActuatorsLinear.Keys);
208 IEnumerable<uint> missingActuatorIdsRotational = actuatorIdsToAddRotational.Except(initialMachine.Entities.ActuatorsRotational.Keys);
209 IEnumerable<uint> missingAxisIdsLinear = axisIdsToAddLinear.Except(initialMachine.Entities.AxesLinear.Keys);
210 IEnumerable<uint> missingAxisIdsRotational = axisIdsToAddRotational.Except(initialMachine.Entities.AxesRotational.Keys);
211
212 if (missingActuatorIdsLinear.Any())
213 return new ArgumentException(
214 $"Cannot add kinematics '{kinematicsToAdd.Name}' because linear axis(s) " +
215 $"({string.Join(", ", missingActuatorIdsLinear)}) must first be assigned to machine '{initialMachine.Name}'.");
216
217 if (missingActuatorIdsRotational.Any())
218 return new ArgumentException(
219 $"Cannot add kinematics '{kinematicsToAdd.Name}' because rotaional axis(s) " +
220 $"({string.Join(", ", missingActuatorIdsRotational)}) must first be assigned to machine '{initialMachine.Name}'.");
221
222 if (missingAxisIdsLinear.Any())
223 return new ArgumentException(
224 $"Cannot add kinematics '{kinematicsToAdd.Name}' because linear axis(s) " +
225 $"({string.Join(", ", missingAxisIdsLinear)}) must first be assigned to machine '{initialMachine.Name}'.");
226
227 if (missingAxisIdsRotational.Any())
228 return new ArgumentException(
229 $"Cannot add kinematics '{kinematicsToAdd.Name}' because rotaional axis(s) " +
230 $"({string.Join(", ", missingAxisIdsRotational)}) must first be assigned to machine '{initialMachine.Name}'.");
231
232 // Ensure endstops have been added to the machine.
233 IEnumerable<uint> allEndstopIdsToAdd = endstopSetsToAdd
234 .SelectMany(set => new[] { set.MinEndstopId, set.MaxEndstopId })
235 .WhereNotNull();
236
237 IEnumerable<uint> missingEndstopIds = allEndstopIdsToAdd.Except(initialMachine.Entities.Endstops.Keys);
238
239 if (missingEndstopIds.Any())
240 return new ArgumentException(
241 $"Cannot add kinematics '{kinematicsToAdd.Name}' because endstop(s) " +
242 $"({string.Join(", ", missingEndstopIds)}) must first be assigned to machine '{initialMachine.Name}'.");
243
244 // Make sure no other kinematic systems use the same axes or actuators.
245 IEnumerable<uint> actuatorIdsInUseLinear = ActuatorsLinear.Select(actuator => actuator.Id);
246 IEnumerable<uint> actuatorIdsInUseRotational = ActuatorsRotational.Select(actuator => actuator.Id);
247 IEnumerable<uint> axisIdsInUseLinear = AxesLinear.Select(axis => axis.Id);
248 IEnumerable<uint> axisIdsInUseRotational = AxesRotational.Select(axis => axis.Id);
249
250 IEnumerable<uint> actuatorIdsUsedTwiceLinear = actuatorIdsToAddLinear.Intersect(actuatorIdsInUseLinear);
251 IEnumerable<uint> actuatorIdsUsedTwiceRotational = actuatorIdsToAddRotational.Intersect(actuatorIdsInUseRotational);
252 IEnumerable<uint> axisIdsUsedTwiceLinear = axisIdsToAddLinear.Intersect(axisIdsInUseLinear);
253 IEnumerable<uint> axisIdsUsedTwiceRotational = axisIdsToAddRotational.Intersect(axisIdsInUseRotational);
254
255 if (actuatorIdsUsedTwiceLinear.Any())
256 return new ArgumentException(
257 $"Cannot add kinematics '{kinematicsToAdd.Name}' because linear axis(s) " +
258 $"({string.Join(", ", actuatorIdsUsedTwiceLinear)}) are already in use by another kinematics.");
259
260 if (actuatorIdsUsedTwiceRotational.Any())
261 return new ArgumentException(
262 $"Cannot add kinematics '{kinematicsToAdd.Name}' because rotational axis(s) " +
263 $"({string.Join(", ", actuatorIdsUsedTwiceRotational)}) are already in use by another kinematics.");
264
265 if (axisIdsUsedTwiceLinear.Any())
266 return new ArgumentException(
267 $"Cannot add kinematics '{kinematicsToAdd.Name}' because linear axis(s) " +
268 $"({string.Join(", ", axisIdsUsedTwiceLinear)}) are already in use by another kinematics.");
269
270 if (axisIdsUsedTwiceRotational.Any())
271 return new ArgumentException(
272 $"Cannot add kinematics '{kinematicsToAdd.Name}' because rotational axis(s) " +
273 $"({string.Join(", ", axisIdsUsedTwiceRotational)}) are already in use by another kinematics.");
274
275 // Validation complete, now work on returning modified copy with additional actuators, axes, and kinematics.
276 // First create new lists of all the actuators and axes.
277 IEnumerable<IActuatorLinear> actuatorsToAddLinear = initialMachine.Entities.ActuatorsLinear.Values
278 .Where(actuator => actuatorIdsToAddLinear.Contains(actuator.Id));
279
280 IEnumerable<IActuatorRotational> actuatorsToAddRotational = initialMachine.Entities.ActuatorsRotational.Values
281 .Where(actuator => actuatorIdsToAddRotational.Contains(actuator.Id));
282
283 IEnumerable<AxisLinear> axesToAddLinear = initialMachine.Entities.AxesLinear.Values
284 .Where(axis => axisIdsToAddLinear.Contains(axis.Id));
285
286 IEnumerable<AxisRotational> axesToAddRotational = initialMachine.Entities.AxesRotational.Values
287 .Where(axis => axisIdsToAddRotational.Contains(axis.Id));
288
289 IImmutableList<IKinematics> kinematics = Kinematics.Add(kinematicsToAdd);
290 IImmutableList<IActuatorLinear> allActuatorsLinear = ActuatorsLinear.AddRange(actuatorsToAddLinear);
291 IImmutableList<IActuatorRotational> allActuatorsRotational = ActuatorsRotational.AddRange(actuatorsToAddRotational);
292 IImmutableList<AxisLinear> allAxesLinear = AxesLinear.AddRange(axesToAddLinear);
293 IImmutableList<AxisRotational> allAxesRotational = AxesRotational.AddRange(axesToAddRotational);
294 IImmutableList<EndstopSet> allEndstopSets = EndstopSets.AddRange(endstopSetsToAdd);
295
296 // Next, initialize arrays to use for calculating forward and inverse kinematics.
297 uint degreesOfFreedom = (uint)kinematics.Sum(k => k.DegreesOfFreedom);
298 uint axisLinearCount = (uint)(AxesLinear.Count + axisIdsToAddLinear.Count);
299 uint axisRotationalCount = (uint)(AxesRotational.Count + axisIdsToAddRotational.Count);
300 uint actuatorLinearCount = (uint)(ActuatorsLinear.Count + actuatorIdsToAddLinear.Count);
301 uint actuatorRotationalCount = (uint)(ActuatorsRotational.Count + actuatorIdsToAddRotational.Count);
302
303 Length?[] actuatorPositionsLinear = new Length?[actuatorLinearCount];
304 Angle?[] actuatorPositionsRotational = new Angle?[actuatorRotationalCount];
305 Length?[] axisPositionsLinear = new Length?[axisLinearCount];
306 Angle?[] axisPositionsRotational = new Angle?[axisRotationalCount];
307 List<Memory<Length?>> actuatorMemoriesLinear = [];
308 List<Memory<Angle?>> actuatorMemoriesRotational = [];
309 List<Memory<Length?>> axisMemoriesLinear = [];
310 List<Memory<Angle?>> axisMemoriesRotational = [];
311
312 int actuatorMemoryIndexLinear = 0;
313 int actuatorMemoryIndexRotational = 0;
314 int axisMemoryIndexLinear = 0;
315 int axisMemoryIndexRotational = 0;
316
317 // Layout the memory for each kinematics contained in the CompositeKinematicSystem.
318 foreach (IKinematics kin in kinematics)
319 {
320 actuatorMemoriesLinear.Add(new(
321 actuatorPositionsLinear,
322 actuatorMemoryIndexLinear,
323 (int)kin.ActuatorLinearCount));
324
325 actuatorMemoryIndexLinear += (int)kin.ActuatorLinearCount;
326
327 actuatorMemoriesRotational.Add(new(
328 actuatorPositionsRotational,
329 actuatorMemoryIndexRotational,
330 (int)kin.ActuatorRotationalCount));
331
332 actuatorMemoryIndexLinear += (int)kin.ActuatorRotationalCount;
333
334 axisMemoriesLinear.Add(new(
335 axisPositionsLinear,
336 axisMemoryIndexLinear,
337 (int)kin.AxisLinearCount));
338
339 axisMemoryIndexLinear += (int)kin.AxisLinearCount;
340
341 axisMemoriesRotational.Add(new(
342 axisPositionsRotational,
343 axisMemoryIndexRotational,
344 (int)kin.AxisRotationalCount));
345
346 axisMemoryIndexRotational += (int)kin.AxisRotationalCount;
347 }
348
349 return new CompositeKinematicSystem(
350 degreesOfFreedom,
351 kinematics,
352 allAxesLinear,
353 allAxesRotational,
354 allActuatorsLinear,
355 allActuatorsRotational,
356 allEndstopSets,
357 actuatorPositionsLinear,
358 actuatorPositionsRotational,
359 axisPositionsLinear,
360 axisPositionsRotational,
361 actuatorMemoriesLinear,
362 actuatorMemoriesRotational,
363 axisMemoriesLinear,
364 axisMemoriesRotational);
365 }
