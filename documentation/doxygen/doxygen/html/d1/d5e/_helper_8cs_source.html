179 public static Result<Duration> TimeOfConstAccelerationMove(Speed vi, Acceleration a, Length l)
180 {
181 if (l.Equals(Length.Zero, Length.FromMillimeters(0.0001)))
182 return Duration.Zero;
183
184 if (a.Equals(Acceleration.Zero, Acceleration.FromMillimetersPerSecondSquared(0.01)))
185 {
186 if (vi.Equals(Speed.Zero, Speed.FromMillimetersPerSecond(0.0001)))
187 return new ArgumentException("Cannot reach destination wtih acceleration and velocity of zero.");
188
189 return l / vi;
190 }
191
192 double viSquared = Math.Pow(vi.MillimetersPerSecond, 2);
193 double aMmPerSec2 = a.MillimetersPerSecondSquared;
194 double squareRoot = Math.Sqrt(viSquared + 2 * aMmPerSec2 * l.Millimeters);
195
196 if (!double.IsNormal(squareRoot))
197 return new ArgumentException("No solution to constant velocity equation");
198
199 double s1 = ((-viSquared + squareRoot) / aMmPerSec2);
200 double s2 = ((-viSquared - squareRoot) / aMmPerSec2);
201 if (s1 > 0)
202 {
203 if (s2 > 0)
204 return Duration.FromSeconds(Math.Min(s1, s2));
205 else
206 return Duration.FromSeconds(s1);
207 }
208 else
209 {
210 if (s2 > 0)
211 return Duration.FromSeconds(s2);
212 else
213 return new ArgumentException("No positive solution to constant velocity equation");
214 }
215 }
