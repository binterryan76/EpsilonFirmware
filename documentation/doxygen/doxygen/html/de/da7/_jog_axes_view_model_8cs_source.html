154 private void MoveAxisLinear(AxisLinear? axis, Length distance)
155 {
156 if (axis is null)
157 return;
158
159 ImmutableDictionary<uint, Length> distances =
160 ImmutableDictionary.CreateRange(new Dictionary<uint, Length>
161 {
162 { axis.Id, distance }
163 });
164
165 Result<MoveCommand> move = MoveCommand.New(
166 linearPositions: distances,
167 absoluteMove: false);
168
169 if (move.IsError)
170 {
171 ResultMessageLogger.Log($"Cannot move: {move.Exception.Message}");
172 return;
173 }
174
175 Engine.EnqueueCommand(MachineId, move.Value);
176 }

261 private void ReadAllAxes(Machine? machine = null)
262 {
263 Axis1 = null;
264 Axis2 = null;
265 Axis3 = null;
266 OtherAxesLinear.Clear();
267 AllAxesRotational.Clear();
268
269 machine ??= Engine.GetCurrentMachine(MachineId);
270
271 if (machine is null)
272 return;
273
274 List<AxisLinear> nonExtrusionAxesLinear =
275 [.. machine.Entities.AxesLinear.Values.Where(axis => !axis.IsExtrusionAxis)];
276
277 for (int i = 0; i < nonExtrusionAxesLinear.Count; i++)
278 {
279 AxisLinear axis = nonExtrusionAxesLinear[i];
280 switch (i)
281 {
282 case 0:
283 Axis1 = axis;
284 break;
285 case 1:
286 Axis2 = axis;
287 break;
288 case 2:
289 Axis3 = axis;
290 break;
291 default:
292 OtherAxesLinear.Add(new AxisLinearButtonsViewModel(axis, !DisableAllButtonsFlag));
293 break;
294 }
295 }
296
297 List<AxisRotational> nonExtrusionAxesRotational =
298 [.. machine.Entities.AxesRotational.Values.Where(axis => !axis.IsExtrusionAxis)];
299
300 for (int i = 0; i < nonExtrusionAxesRotational.Count; i++)
301 AllAxesRotational.Add(new AxisRotationalButtonsViewModel(nonExtrusionAxesRotational[i], !DisableAllButtonsFlag));
302
303 SetButtonEnabledAndVisibility();
304 }

328 private void SetButtonEnabledAndVisibility()
329 {
330 // First set the visibility of the axes based on whether they are null or not.
331 Axis1Visibility = Axis1 is not null ? Visibility.Visible : Visibility.Collapsed;
332 Axis2Visibility = Axis2 is not null ? Visibility.Visible : Visibility.Collapsed;
333 Axis3Visibility = Axis3 is not null ? Visibility.Visible : Visibility.Collapsed;
334
335 // Axis1 is the first linear axis to be set so that is all we need to check to determine if a linear axis is present.
336 bool atLeastOneLinearAxisPresent = Axis1 is not null;
337 bool atLeastOneRotationalAxisPresent = AllAxesRotational.Count > 0;
338 bool atLeastOneAxisPresent = atLeastOneLinearAxisPresent || atLeastOneRotationalAxisPresent;
339
340 // One axis must exist to home all.
341 HomeAllVisibility = atLeastOneAxisPresent ? Visibility.Visible : Visibility.Collapsed;
342
343 // If the DisableAllButtonsFlag is set, disable all buttons and return early.
344 if (DisableAllButtonsFlag)
345 {
346 Axis1Enabled = false;
347 Axis2Enabled = false;
348 Axis3Enabled = false;
349 HomeAllEnabled = false;
350
351 foreach (AxisLinearButtonsViewModel linearAxisButtons in OtherAxesLinear)
352 linearAxisButtons.ButtonsEnabled = false;
353
354 foreach (AxisRotationalButtonsViewModel rotationalAxisButtons in AllAxesRotational)
355 rotationalAxisButtons.ButtonsEnabled = false;
356
357 return;
358 }
359
360 Axis1Enabled = Axis1 is not null;
361 Axis2Enabled = Axis2 is not null;
362 Axis3Enabled = Axis3 is not null;
363
364 // Axes will never be null so enable all buttons for the other axes.
365 foreach (AxisLinearButtonsViewModel linearAxisButtons in OtherAxesLinear)
366 linearAxisButtons.ButtonsEnabled = true;
367
368 foreach (AxisRotationalButtonsViewModel rotationalAxisButtons in AllAxesRotational)
369 rotationalAxisButtons.ButtonsEnabled = true;
370
371 bool enableHomeAll = ShouldEnableHomeAll();
372
373 bool ShouldEnableHomeAll()
374 {
375 if (Axis1 is not null && !Axis1Enabled)
376 return false;
377
378 if (Axis2 is not null && !Axis2Enabled)
379 return false;
380
381 if (Axis3 is not null && !Axis3Enabled)
382 return false;
383
384 if (OtherAxesLinear.Any(axisButtons => !axisButtons.ButtonsEnabled))
385 return false;
386
387 if (AllAxesRotational.Any(axisButtons => !axisButtons.ButtonsEnabled))
388 return false;
389
390 return true;
391 }
392 }
