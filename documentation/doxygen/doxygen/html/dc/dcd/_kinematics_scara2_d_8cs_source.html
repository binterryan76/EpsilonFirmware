95 public (Angle angleProximal, Angle angleDistal) InverseKinematics(Length x, Length y)
96 {
97 double numerator = Math.Pow(x.Millimeters, 2) + Math.Pow(y.Millimeters, 2) - SumSquaredLengths;
98 double denominator = 2 * L1L2;
99 double ratio = numerator / denominator;
100
101 if (ratio > 1 || ratio < -1)
102 throw new ArgumentException($"X and Y are outside workspace. X: {x}, Y: {y}");
103
104 Angle angleDistal = Angle.FromRadians(Math.Acos(ratio));
105
106 double part1 = Math.Atan2(y.Millimeters, x.Millimeters);
107 double part2 = Math.Atan2(LengthDistalMm * Math.Sin(angleDistal.Radians), LengthProximal.Millimeters + LengthDistalMm * Math.Cos(angleDistal.Radians));
108 Angle angleProximal = Angle.FromRadians(part1 - part2);
109 return (angleProximal, angleDistal);

119 public (RotationalSpeed speedProximal, RotationalSpeed speedDistal) InverseSpeeds(Speed velocityX, Speed velocityY, Angle angleProximal)
120 {
121 if (velocityX.Equals(Speed.Zero, Speed.FromMillimetersPerSecond(0.01)) && velocityY.Equals(Speed.Zero, Speed.FromMillimetersPerSecond(0.01)))
122 return (RotationalSpeed.Zero, RotationalSpeed.Zero);
123
124 double vx = velocityX.MillimetersPerSecond;
125 double vy = velocityY.MillimetersPerSecond;
126 double sumVelocitiesSquared = Math.Pow(vx, 2) + Math.Pow(vy, 2);
127
128 // these are m^4
129 double numerator = Math.Pow(-SumSquaredLengths + sumVelocitiesSquared, 2);
130 double denominator = 4 * L1SquaredL2Squared;
131
132 // this is m^2
133 double fullDenominator = L1L2 * Math.Sqrt(1 - (numerator / denominator));
134
135 // this is in Angle per meter
136 double dThetaDx = -vx / fullDenominator;
137 double dThetaDy = -vy / fullDenominator;
138
139 // dθ/dt = dθ/dx * dx/dt + dθ/dy * dy/dt
140 RotationalSpeed speedDistal = RotationalSpeed.FromRadiansPerSecond(dThetaDx * vx + dThetaDy * vy);
141
142 double dTheta2Dx = -vy / sumVelocitiesSquared;
143 double dTheta2Dy = vx / sumVelocitiesSquared;
144
145 double cos = Math.Cos(angleProximal.Radians);
146 double sin = Math.Sin(angleProximal.Radians);
147 double cos2 = Math.Pow(cos, 2);
148 double sin2 = Math.Pow(sin, 2);
149 double sum = L2Squared * sin2 + L2Squared * cos2;
150 double product = L1L2 * cos;
151 double numerator2 = product + sum;
152 double denominator2 = L1Squared + (2 * product) + sum;
153 double dTheta2Dtheta = -numerator2 / denominator2;
154
155 // dθ2/dt = dθ2/dx * dx/dt + dθ2/dy * dy/dt + dθ2/dθ * dθ/dt
156 RotationalSpeed speedProximal =
157 RotationalSpeed.FromRadiansPerSecond(dTheta2Dx * vx + dTheta2Dy * vy) +
158 dTheta2Dtheta * speedDistal;
159
160 return (speedProximal, speedDistal);
161 }
