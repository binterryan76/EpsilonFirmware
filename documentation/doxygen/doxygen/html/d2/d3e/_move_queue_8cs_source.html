154 private static (Frequency previousMoveEndDsDtLimit, Frequency currentMoveStartDsDtLimit) CalculateJunctionDsDtLimits(
155 MovePoint previousMoveEndPoint,
156 MovePoint currentMoveStartPoint,
157 CompositeKinematicSystem previousMoveKinematicSystem,
158 CompositeKinematicSystem currentMoveKinematicSystem)
159 {
160 double cosineOfTurnAngle = ComputeCosineOfTurnAngle(
161 previousMoveEndPoint,
162 currentMoveStartPoint,
163 previousMoveKinematicSystem,
164 currentMoveKinematicSystem);
165 double cornerSpeedFactor = ComputeCornerSpeedFactor(cosineOfTurnAngle);
166
167 Frequency previousMoveEndDsDtLimit = Constants.LARGEST_FREQUENCY;
168 Frequency currentMoveStartDsDtLimit = Constants.LARGEST_FREQUENCY;
169
170 for (int i = 0; i < previousMoveKinematicSystem.ActuatorsLinear.Count; i++)
171 {
172 Speed actuatorJunctionVelocity =
173 previousMoveKinematicSystem.ActuatorsLinear[i].SquareCornerSpeed * cornerSpeedFactor;
174
175 Length derivativeMagnitude = previousMoveEndPoint.ActuatorPositionDerivatives1Linear[i].FastAbs();
176
177 // Dividing this actuator's own junction velocity by
178 // this same actuator's own dTheta/ds (units of Length/Angle)
179 // gives ds/dt (units of frequency) which can be compared between
180 // linear and rotational actuators.
181 if (derivativeMagnitude > Constants.VERY_SMALL_LENGTH)
182 {
183 Frequency candidatePathDsDtLimit = actuatorJunctionVelocity / derivativeMagnitude;
184 previousMoveEndDsDtLimit =
185 UnitsNetHelpers.Helpers.Min(previousMoveEndDsDtLimit, candidatePathDsDtLimit);
186 }
187 }
188
189 for (int i = 0; i < previousMoveKinematicSystem.ActuatorsRotational.Count; i++)
190 {
191 RotationalSpeed actuatorJunctionVelocity =
192 previousMoveKinematicSystem.ActuatorsRotational[i].SquareCornerSpeed * cornerSpeedFactor;
193
194 Angle derivativeMagnitude = previousMoveEndPoint.ActuatorPositionDerivatives1Rotational[i].FastAbs();
195
196 if (derivativeMagnitude > Constants.VERY_SMALL_ANGLE)
197 {
198 Frequency candidatePathDsDtLimit = actuatorJunctionVelocity / derivativeMagnitude;
199 previousMoveEndDsDtLimit =
200 UnitsNetHelpers.Helpers.Min(previousMoveEndDsDtLimit, candidatePathDsDtLimit);
201 }
202 }
203
204 for (int i = 0; i < currentMoveKinematicSystem.ActuatorsLinear.Count; i++)
205 {
206 Speed actuatorJunctionVelocity =
207 currentMoveKinematicSystem.ActuatorsLinear[i].SquareCornerSpeed * cornerSpeedFactor;
208
209 Length derivativeMagnitude = currentMoveStartPoint.ActuatorPositionDerivatives1Linear[i].FastAbs();
210
211 if (derivativeMagnitude > Constants.VERY_SMALL_LENGTH)
212 {
213 Frequency candidatePathDsDtLimit = actuatorJunctionVelocity / derivativeMagnitude;
214 currentMoveStartDsDtLimit =
215 UnitsNetHelpers.Helpers.Min(currentMoveStartDsDtLimit, candidatePathDsDtLimit);
216 }
217 }
218
219 for (int i = 0; i < currentMoveKinematicSystem.ActuatorsRotational.Count; i++)
220 {
221 RotationalSpeed actuatorJunctionVelocity =
222 currentMoveKinematicSystem.ActuatorsRotational[i].SquareCornerSpeed * cornerSpeedFactor;
223
224 Angle derivativeMagnitude = currentMoveStartPoint.ActuatorPositionDerivatives1Rotational[i].FastAbs();
225
226 if (derivativeMagnitude > Constants.VERY_SMALL_ANGLE)
227 {
228 Frequency candidatePathDsDtLimit = actuatorJunctionVelocity / derivativeMagnitude;
229 currentMoveStartDsDtLimit =
230 UnitsNetHelpers.Helpers.Min(currentMoveStartDsDtLimit, candidatePathDsDtLimit);
231 }
232 }
233
234 return (previousMoveEndDsDtLimit, currentMoveStartDsDtLimit);
235 }

247 private static double ComputeCosineOfTurnAngle(
248 MovePoint previousMoveEndPoint,
249 MovePoint currentMoveStartPoint,
250 CompositeKinematicSystem previousMoveKinematicSystem,
251 CompositeKinematicSystem currentMoveKinematicSystem)
252 {
253 // normalize the dƟ/ds derivatives by dividing by square corner speed so that linear and rotational
254 // axes can be compared in the same units.
255 List<Duration?> normalizedDerivativesPrevious =
256 previousMoveEndPoint.NormalizedActuatorDerivatives(previousMoveKinematicSystem);
257 List<Duration?> normalizedDerivativesCurrent =
258 currentMoveStartPoint.NormalizedActuatorDerivatives(currentMoveKinematicSystem);
259
260 // Get the magnitude of the normalized derivatives so we can calculate the angle between them.
261 Duration previousEndDerivativeMagnitue =
262 UnitsNetHelpers.Helpers.Distance(normalizedDerivativesPrevious);
263 Duration currentStartDerivativeMagnitue =
264 UnitsNetHelpers.Helpers.Distance(normalizedDerivativesCurrent);
265
266 // Note: The angle between two vectors (Ɵ) can be calculated with this formula:
267 // Cos(Ɵ) = Dot(VectorA, VectorB) / (|VectorA| * |VectorB|)
268 // The dot product is a scalar equal to the sum of products of the corresponding components of two vectors.
269
270 if (previousEndDerivativeMagnitue < Constants.VERY_SMALL_DURATION ||
271 currentStartDerivativeMagnitue < Constants.VERY_SMALL_DURATION)
272 {
273 // One side has a degenerate tangent (an actuator momentarily stationary
274 // relative to s right at the junction). Treat this as a full reversal
275 // (must reach zero velocity) so we stay on the safe side rather than
276 // dividing by a near-zero magnitude.
277 return -1.0;
278 }
279
280 double dotProductSecondsSquared = normalizedDerivativesPrevious
281 .Zip(normalizedDerivativesCurrent,
282 (a, b) => a.HasValue && b.HasValue ? a.Value.Seconds * b.Value.Seconds : 0.0)
283 .Sum();
284 double cosineOfTurnAngle = dotProductSecondsSquared /
285 (previousEndDerivativeMagnitue.Seconds * currentStartDerivativeMagnitue.Seconds);
286
287 return Math.Clamp(cosineOfTurnAngle, -1.0, 1.0);
288 }

300 private static double ComputeCornerSpeedFactor(double cosineOfTurnAngle)
301 {
302 // cos(turnAngle/2) is calculated using the half-angle identity, which lets us avoid
303 // calculating an inverse cosine which is a little more costly.
304 // half-angle identity: cos(x/2) = sqrt((1 + cos(x)) / 2) for x in [0, pi].
305 double cosineOfHalfTurnAngle = Math.Sqrt((1.0 + cosineOfTurnAngle) / 2.0);
306 double denominator = 1.0 - cosineOfHalfTurnAngle;
307
308 // Turn angle is essentially zero (straight through): no junction limit applies.
309 // Returning a very large factor ensures that the junction speed limits won't be a
310 // limiting factor because this value goes through a min function.
311 if (denominator < Constants.VERY_SMALL_NUMBER)
312 return double.PositiveInfinity;
313
314 double squareRootTerm = Constants.ROOT_2_MINUS_1 * cosineOfHalfTurnAngle / denominator;
315 return Math.Sqrt(squareRootTerm);
316 }

325 public void TryWriteToCsv(
326 string filePath,
327 CompositeKinematicSystem kinematicSystem,
328 bool writeSimplifiedPoints)
329 {
330 StringBuilder output = new();
331
332 // Make column headers.
333 IEnumerable<string> headers = new List<string>(["Time (s)"])
334 .Concat(kinematicSystem.AxesLinear.Select(axis => axis.Name + " Pos (mm)"))
335 .Concat(kinematicSystem.AxesRotational.Select(axis => axis.Name + " Pos (deg)"))
336 .Concat(kinematicSystem.ActuatorsLinear.Select(actuator => actuator.Name + " Pos (mm)"))
337 .Concat(kinematicSystem.ActuatorsRotational.Select(actuator => actuator.Name + " Pos (deg)"))
338 .Concat(kinematicSystem.ActuatorsLinear.Select(actuator => actuator.Name + " Speed (mm/s)"))
339 .Concat(kinematicSystem.ActuatorsRotational.Select(actuator => actuator.Name + " Speed (deg/s)"));
340
341 output.AppendLine(string.Join(",", headers));
342
343 Duration timeOfStartOfMove = Duration.Zero;
344
345 foreach (Move move in queue)
346 {
347 IEnumerable<MovePoint> pointsToWrite = writeSimplifiedPoints ?
348 move.SimplifiedPointIndicies.Select(i => move.Points[i]) :
349 move.Points;
350
351 foreach (MovePoint point in pointsToWrite)
352 {
353 IEnumerable<string> values = new List<string>([(timeOfStartOfMove + point.Time).Seconds.ToString("F6")])
354 .Concat(point.AxisPositionsLinear.Select(pos => pos.HasValue ? pos.Value.Millimeters.ToString("F2") : string.Empty))
355 .Concat(point.AxisPositionsRotational.Select(pos => pos.HasValue ? pos.Value.Degrees.ToString("F2") : string.Empty))
356 .Concat(point.ActuatorPositionsLinear.Select(pos => pos.HasValue ? pos.Value.Millimeters.ToString("F2") : string.Empty))
357 .Concat(point.ActuatorPositionsRotational.Select(pos => pos.HasValue ? pos.Value.Degrees.ToString("F2") : string.Empty))
358 .Concat(point.ActuatorPositionDerivatives1Linear.Select(derivative =>
359 derivative.HasValue ? (derivative.Value * point.ActualDsDt).MillimetersPerSecond.ToString("F2") : string.Empty))
360 .Concat(point.ActuatorPositionDerivatives1Rotational.Select(derivative =>
361 derivative.HasValue ? (derivative.Value * point.ActualDsDt).DegreesPerSecond.ToString("F2") : string.Empty));
362 output.AppendLine(string.Join(",", values));
363 }
364 timeOfStartOfMove += move.Points[^1].Time;
365 }
366
367 try
368 {
369 File.WriteAllText(filePath, output.ToString());
370 }
371 catch
372 {
373 // intentionally blank
374 }
375 }
