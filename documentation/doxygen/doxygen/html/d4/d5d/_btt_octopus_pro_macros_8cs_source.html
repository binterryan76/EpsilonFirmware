234 public static void AddBttOctopusProBoardAndPinsMacro(
235 EpsilonEngine engine,
236 uint machineIndex,
237 string boardName,
238 uint communicatorId,
239 uint? lineNumber = null)
240 {
241 Result<BoardTimer> mainBoardTimerLowPrecision = BoardTimer.New(
242 "STM32 TIM 5 Low Precision Timer",
243 Duration.FromMilliseconds(UInt32.MaxValue),
244 Duration.FromMilliseconds(1));
245
246 Result<BoardTimer> mainBoardTimerMotor = BoardTimer.New(
247 "STM32 TIM 4/3 Stepper Motor Timer",
248 Duration.FromMicroseconds(UInt16.MaxValue),
249 Duration.FromMicroseconds(1));
250
251 Debug.Assert(mainBoardTimerLowPrecision.IsSuccess);
252 Debug.Assert(mainBoardTimerMotor.IsSuccess);
253
254 Board mainBoard = new(
255 boardName,
256 communicatorId,
257 Frequency.FromMegahertz(180),
258 mainBoardTimerLowPrecision.Value,
259 mainBoardTimerMotor.Value,
260 Board.IntSize.Bits32,
261 Board.FloatSize.Bits32,
262 isLittleEndian: true);
263
264 engine.EnqueueCommand(machineIndex, new AddBoardCommand(mainBoard, lineNumber++));
265
266 // generate pin names PA0 - PG15
267 IEnumerable<string> pinNames = Stm32.GeneratePinNames('G', 15);
268 byte boardsPinId = 0;
269 foreach (string pinName in pinNames)
270 {
271 Pin pin = new(
272 boardsPinId,
273 pinName,
274 boardId: 0,
275 isOutput: BttOctopusProInputPinNames.Contains(pinName),
276 isDigital: true);
277
278 engine.EnqueueCommand(machineIndex, new AddPinCommand(pin, lineNumber++));
279
280 boardsPinId++;
281 }
282 }
