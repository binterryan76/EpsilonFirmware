117 {

317 private void EnqueueNextCommand(MachineQueue machineQueue)
318 {
319 if (machineQueue.EnqueueStatus != MachineQueue.MachineQueueStatus.Running)
320 return;
321
322 if (machineQueue.CommandsToEnqueue.IsEmpty)
323 {
324 if (machineQueue.Queued.Count <= 0)
325 return;
326
327 SolveMovesAndMarkCommandsAsReadyToSend(machineQueue);
328 return;
329 }
330
331 bool success = machineQueue.CommandsToEnqueue.TryDequeue(out ICommand? command);
332
333 Debug.Assert(success,
334 $"Dequeueing {nameof(MachineQueue.CommandsToEnqueue)} should never fail.");
335
336 Debug.Assert(command is not null,
337 $"{nameof(MachineQueue.CommandsToEnqueue)} should never contain a null command.");
338
339 Machine initialMachine = machineQueue.LatestQueuedMachine;
340 QueuedCommand queuedCommand = command.EnqueueCommandSpecific(initialMachine);
341
342 // check for explicit enqueue command failures
343 if (queuedCommand.ErrorLevel != ErrorLevel.Success)
344 {
345 // TODO: raise event for failed machineQueue event
346
347 // Log error.
348 initialMachine.ResultMessageLogger.Log(Helper.GetFormattedDisplayMessage(queuedCommand));
349
350 // Don't send any more commands to this machineQueue until error is addressed.
351 machineQueue.CommandsToEnqueue.Clear();
352 machineQueue.EnqueueStatus = MachineQueue.MachineQueueStatus.Stopped;
353 return;
354 }
355
356 Debug.Assert(queuedCommand.ResultantMachine is not null,
357 "All success results require a non-null ResultantMachine property.");
358
359 machineQueue.LatestQueuedMachine = queuedCommand.ResultantMachine;
360
361 // If a non-move command comes in and there is nothing before it then just mark it as ready to send.
362 if (machineQueue.Queued.Count <= 0 && command is not MoveCommand)
363 {
364 machineQueue.AddCommandToReadyToSend(queuedCommand);
365 return;
366 }
367
368 // Otherwise, mark it as only queued but not ready to send.
369 machineQueue.Queued.Enqueue(queuedCommand);
370 CommandQueued?.Invoke(initialMachine, queuedCommand);
371
372 if (command is MoveCommand moveCommand)
373 {
374 Result<Move> move = moveCommand.GetMove(initialMachine);
375 if (move.IsError)
376 {
377 // Log error.
378 initialMachine.ResultMessageLogger.Log(move.Exception.Message);
379
380 // Don't send any more commands to this machineQueue until error is addressed.
381 machineQueue.CommandsToEnqueue.Clear();
382 machineQueue.EnqueueStatus = MachineQueue.MachineQueueStatus.Stopped;
383
384 // Go ahead and solve and send all the moves ahead of this command that did work.
385 SolveMovesAndMarkCommandsAsReadyToSend(machineQueue);
386 return;
387 }
388 machineQueue.MoveQueue.Add(move.Value);
389 }
390
391 const int LOOK_AHEAD_COUNT = 20;
392
393 if (command.RequiresZeroVelocity || machineQueue.CommandsToEnqueue.IsEmpty)
394 {
395 // All moves can be solved and sent if the command requires zero
396 // velocity because solving moves will end with zero velocity.
397 // Also if there are no more commands to send then we can solve the moves and send them.
398 SolveMovesAndMarkCommandsAsReadyToSend(machineQueue);
399 }
400 else if (machineQueue.MoveQueue.Count > LOOK_AHEAD_COUNT)
401 {
402 // Enough commands have been queued to the move queue to solve the moves
403 // and add them to the command machineQueue.
404 // We will only send half of them though because if more moves are coming up, we may
405 // want to maintain a higher speed and solvig moves always ends with zero velocity.
406 // Roughly the first half will be accelerating and the second half will
407 // be decelerating but we don't know if we want to decelerate yet.
408 SolveMovesAndMarkHalfOfMoveCommandsAsReadyToSend(machineQueue);
409 }

417 private static void SolveMovesAndMarkCommandsAsReadyToSend(MachineQueue machineQueue)
418 {
419 MoveQueue moveQueue = machineQueue.MoveQueue;
420 moveQueue.SolveAllMoves(machineQueue.LatestQueuedMachine.MotionSystem.Precisions);
421
422 foreach (QueuedCommand command in machineQueue.Queued)
423 machineQueue.AddCommandToReadyToSend(command);
424
425 // Clear the move machineQueue and move commands list but keep the last move to remember velocities for next batch of moves.
426 Move? lastMove = moveQueue.Last;
427 moveQueue.Clear();
428 if (lastMove is not null)
429 moveQueue.Add(lastMove);
430 machineQueue.Queued.Clear();
431 }

439 private static void SolveMovesAndMarkHalfOfMoveCommandsAsReadyToSend(MachineQueue machineQueue)
440 {
441 MoveQueue moveQueue = machineQueue.MoveQueue;
442 moveQueue.SolveAllMoves(machineQueue.LatestQueuedMachine.MotionSystem.Precisions);
443
444 int movesToSend = (moveQueue.Count / 2) - 1;
445 int movesMarkedReadyToSend = 0;
446 int commandsMarkedAsReadyToSend = 0;
447 bool markedEnoughMoves = false;
448 foreach (QueuedCommand queuedCommand in machineQueue.Queued)
449 {
450 bool isMoveCommand = queuedCommand.Command is MoveCommand;
451 if (markedEnoughMoves && isMoveCommand)
452 break;
453
454 machineQueue.AddCommandToReadyToSend(queuedCommand);
455 commandsMarkedAsReadyToSend++;
456
457 if (isMoveCommand)
458 {
459 movesMarkedReadyToSend++;
460 moveQueue.Dequeue();
461 }
462
463 if (movesMarkedReadyToSend >= movesToSend)
464 markedEnoughMoves = true;
465 }
466
467 // Clear all commands that have been sent.
468 machineQueue.Queued = new Queue<QueuedCommand>(machineQueue.Queued.Skip(commandsMarkedAsReadyToSend));
469 }

478 private void SendSomeQueuedCommands(MachineQueue machineQueue)
479 {
480 while (machineQueue.SendStatus == MachineQueue.MachineQueueStatus.Running &&
481 machineQueue.ReadyToSend.Count > 0)
482 {
483 QueuedCommand nextCommand = machineQueue.ReadyToSend.Peek();
484
485 Debug.Assert(nextCommand.ErrorLevel == ErrorLevel.Warning ||
486 nextCommand.ErrorLevel == ErrorLevel.Success,
487 $"{nameof(MachineQueue)}.{nameof(MachineQueue.ReadyToSend)} should only contain {nameof(QueuedCommand)}s with a status of either {ErrorLevel.Warning.ToDisplayStr()} or {ErrorLevel.Success.ToDisplayStr()}.");
488
489 Debug.Assert(nextCommand.ResultantMachine is not null,
490 $"{nameof(MachineQueue)}.{nameof(MachineQueue.ReadyToSend)} should never contain a null {nameof(QueuedCommand)}.{nameof(QueuedCommand.ResultantMachine)}.");
491
492 // Early continue if there is no data to send
493 if (nextCommand.DataPackets.Count <= 0)
494 {
495 // Actually remove the command from the ready to send machineQueue
496 machineQueue.ReadyToSend.Dequeue();
497
498 if (machineQueue.Sent.Count <= 0)
499 {
500 // Just skip the sent list entirely and just apply the resultant machine if
501 // there is no data to send and no commands in front of the current one.
502 machineQueue.CurrentMachine = nextCommand.ResultantMachine;
503
504 // Raise event for command resolved.
505 CommandResolved?.Invoke(nextCommand.ResultantMachine, nextCommand);
506 }
507 else
508 {
509 // We can't apply the resultant machine until the earlier sent commands are completed.
510 machineQueue.Sent.Enqueue(nextCommand);
511 }
512
513 continue;
514 }
515
516 foreach (DataPacket dataPacket in machineQueue.DataPacketsReadyToSend)
517 {
518 uint communicatorId = dataPacket.CommunicatorId;
519 bool success = machineQueue.Communicators.TryGetValue(communicatorId, out ICommunicator? communicatorToSendWith);
520 int dataPacketSize = dataPacket.TotalLength;
521
522 Debug.Assert(success && communicatorToSendWith is not null,
523 $"{nameof(ICommunicator)} {communicatorId} could not be found but should have been verified to exist when the command was queued.");
524
525 if (communicatorToSendWith.BoardCommandBufferBytesFree < dataPacketSize)
526 {
527 // No more room in microcontroller's buffer to send data packet, just
528 // bail without dequeueing the command from machineQueue.ReadyToSend or
529 // dequeueing data packet from machineQueue.DataPacketsReadyToSend.
530 return;
531 }
532
533 // Only assign the sequence number right before sending data packet because if a
534 // command is cancelled, we don't want to increment the sequence number.
535 DataPacket dataPacketWithSequenceNum = dataPacket.SetSequenceNumber(
536 communicatorToSendWith.IncrementAndReturnNextSequenceNumber());
537
538 // Actually send data packet via the ICommunicator.
539 bool sendSuccess = communicatorToSendWith.Send(dataPacketWithSequenceNum);
540
541 IResultMessageLogger logger = nextCommand.ResultantMachine.ResultMessageLogger;
542 if (sendSuccess)
543 {
544 // track how much space the sent command is taking up in the microcontroller's command machineQueue
545 communicatorToSendWith.BoardCommandBufferBytesOccupied += (uint)dataPacketSize;
546
547 // Actually remove the data packet from the ready to send list.
548 machineQueue.DataPacketsReadyToSend.Dequeue();
549
550 // Actually move the command from machineQueue.ReadyToSend to machineQueue.Sent if
551 // every data packet associated with it has been sent successfully.
552 // NOTE: The command that is moved will not have any of the sequence numbers assigned to the data packets
553 // because a modified copy of the data packet is sent to the microcontroller but the data packet in the command's
554 // data packet list is not modified. I don't think we will need to reference the sequence number later so this is ok.
555 if (machineQueue.DataPacketsReadyToSend.Count <= 0)
556 {
557 machineQueue.ReadyToSend.Dequeue();
558 machineQueue.Sent.Enqueue(nextCommand);
559 CommandSent?.Invoke(nextCommand.InitialMachine, nextCommand);
560 }
561 }
562 else
563 {
564 // TODO: Raise event for command send failure.
565
566 // Don't remove the command from machineQueue.ReadyToSend or
567 // the data packet from machineQueue.DataPacketsReadyToSend.
568
569 // log fail
570 logger.Log(
571 Helper.GetFormattedDisplayMessage(
572 nextCommand.Command,
573 ErrorLevel.Error,
574 $"Failed to send data packet using communicator '{communicatorToSendWith.Name}'"));
575
576 // stop machineQueue
577 machineQueue.SendStatus = MachineQueue.MachineQueueStatus.Stopped;
578 }
579 }
580 }

583 private void ProcessDataPacket(ICommunicator communicator, DataPacket dataPacket)
584 {
585 bool success = MachineQueues.TryGetValue(communicator.MachineId, out MachineQueue? machineQueue);
586 if (!success || machineQueue is null)
587 {
588 EngineLogger.Log($"Data packet received from communicator '{communicator.Name}' but MachineQueue {communicator.MachineId} doesn't exist.");
589 return;
590 }
591 Machine machine = machineQueue.CurrentMachine;
592 IResultMessageLogger logger = machine.ResultMessageLogger;
593
594 success = machine.Entities.Boards.TryGetValue(communicator.BoardId, out Board? board);
595 string dataPacketSource;
596 if (success && board is not null)
597 dataPacketSource = $"{machine.Name} (board {board.Id})";
598 else
599 dataPacketSource = $"{machine.Name} (communicator {communicator.Id})";
600
601 Result<CommandCode> commandCode = dataPacket.CommandCode.ToCommandCode();
602
603 // If unrecognized command code is received, log and continue
604 if (commandCode.IsError)
605 {
606 logger.Log($"Unregocnized command code received from {dataPacketSource}: {dataPacket.CommandCode}.");
607 logger.Log(commandCode.Exception.Message);
608 return;
609 }
610
611 // TODO: add validation on data length for each command code type
612 switch (commandCode.Value)
613 {
614 case CommandCode.DISPLAY_TEXT:
615 {
616 string message = System.Text.Encoding.ASCII.GetString(dataPacket.AllData.ToArray()[3..]);
617 logger.Log($"Message from {dataPacketSource}:\n\t{message}.");
618 break;
619 }
620 case CommandCode.COMMAND_COMPLETE:
621 {
622 byte sequenceNum = dataPacket.SequenceNumber;
623
624 if (machineQueue.Sent.Count <= 0)
625 {
626 logger.Log($"Warning: Command {sequenceNum} completed on {dataPacketSource} but there were no commands in MachineQueue.Sent.");
627 return;
628 }
629
630 // remove command from sent machineQueue
631 // TODO: for now just assume that there is only one ICommunicator and Board per machine and so one commands should complete in the order they were sent. Add support for multiple ICommunicators later.
632 // TODO: IMPORTANT: only remove the command from the queue and reduce the space occupied in the buffer(s) if all data packets of the command are complete.
633 QueuedCommand sentCommandInMachineQueue = machineQueue.Sent.Dequeue();
634
635 Debug.Assert(sentCommandInMachineQueue.ResultantMachine is not null,
636 $"A sent command should never have a null {nameof(QueuedCommand.ResultantMachine)}.");
637
638 // Apply resultant machine.
639 machineQueue.CurrentMachine = sentCommandInMachineQueue.ResultantMachine;
640
641 // Commands with data packets should have had data sent to a board via an
642 // ICommunicator and thus should be the next item in the BoardQueue.
643 if (sentCommandInMachineQueue.DataPackets is not null)
644 {
645 if (machineQueue.Sent.Count <= 0)
646 {
647 logger.Log($"Warning: Command {sequenceNum} completed on {dataPacketSource} but command {sequenceNum} wasn't in the sent MachineQueue.");
648 return;
649 }
650
651 if (machineQueue.Sent.Peek() != sentCommandInMachineQueue)
652 {
653 logger.Log($"Warning: Command {sequenceNum} completed on {dataPacketSource} but that command wasn't the next command in the sent MachineQueue.");
654 return;
655 }
656
657 // remove the command from the BoardQueue and reduce the buffer bytes occupied
658 // by the data packet length now that the microcontroller is done with that data packet.
659 foreach (DataPacket resolvedCommandDataPacket in sentCommandInMachineQueue.DataPackets)
660 {
661 success = machineQueue.Communicators.TryGetValue(resolvedCommandDataPacket.CommunicatorId, out ICommunicator? dataPacketsCommunicator);
662 if (!success || dataPacketsCommunicator is null)
663 {
664 logger.Log($"Warning: Command {sequenceNum} completed on {dataPacketSource} but the communicator {resolvedCommandDataPacket.CommunicatorId} for that command's data packet could not be found.");
665 continue;
666 }
667 dataPacketsCommunicator.BoardCommandBufferBytesOccupied -= (uint)resolvedCommandDataPacket.TotalLength;
668 }
669
670 if (sentCommandInMachineQueue.DataPackets.Count > 1)
671 throw new NotImplementedException("Currently commands can only have one data packet. Support for multiple data packets per command is not yet implemented.");
672
673 CommandResolved?.Invoke(sentCommandInMachineQueue.ResultantMachine, sentCommandInMachineQueue);
674 }
675
676 break;
677 }
678 case CommandCode.ERROR:
679 {
680 // log errors from microcontroller
681 // TODO: require board restart?
682 byte[] allData = [.. dataPacket.AllData];
683 Result<BoardErrorCode> errorCode = allData[3].ToBoardErrorCode();
684 if (errorCode.IsError)
685 {
686 logger.Log($"Unrecognized error with command {dataPacket.SequenceNumber} on {dataPacketSource}.");
687 logger.Log(errorCode.Exception.Message);
688 return;
689 }
690 UInt32 lineNumber = BitConverter.ToUInt32(allData, 4);
691 string fileName = BitConverter.ToString(allData, 4 + sizeof(UInt32));
692 logger.Log($"Error with command {dataPacket.SequenceNumber} on {dataPacketSource}:\n\t{fileName} line {lineNumber}\n\t{errorCode.Value.DisplayMessage()}");
693
694 break;
695 }
696 }
