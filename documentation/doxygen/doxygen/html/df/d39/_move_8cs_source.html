135 public static Result<Move> New(
136 MotionSystemPrecisions precisions,
137 CompositeKinematicSystem kinematicSystem,
138 IList<Length?> initialAxisPositionsLinear,
139 IList<Angle?> initialAxisPositionsRotational,
140 IList<Length?> finalAxisPositionsLinear,
141 IList<Angle?> finalAxisPositionsRotational,
142 Speed? requestedSpeedLinear,
143 RotationalSpeed? requestedSpeedRotational)
144 {
145 if (initialAxisPositionsLinear.Count != kinematicSystem.AxesLinear.Count)
146 return new ArgumentException(
147 $"The kinematics require {kinematicSystem.AxesLinear.Count} initial linear axis positions but {initialAxisPositionsLinear.Count} were provided.");
148
149 if (initialAxisPositionsRotational.Count != kinematicSystem.AxesRotational.Count)
150 return new ArgumentException(
151 $"The kinematics require {kinematicSystem.AxesRotational.Count} initial rotational axis positions but {initialAxisPositionsRotational.Count} were provided.");
152
153 if (finalAxisPositionsLinear.Count != kinematicSystem.AxesLinear.Count)
154 return new ArgumentException(
155 $"The kinematics require {kinematicSystem.AxesLinear.Count} final linear axis positions but {finalAxisPositionsLinear.Count} were provided.");
156
157 if (finalAxisPositionsRotational.Count != kinematicSystem.AxesRotational.Count)
158 return new ArgumentException(
159 $"The kinematics require {kinematicSystem.AxesRotational.Count} final rotational axis positions but {finalAxisPositionsRotational.Count} were provided.");
160
161 if (requestedSpeedLinear is null && requestedSpeedRotational is null)
162 return new ArgumentException(
163 $"Moves require either a linear speed or rotational speed but none were provided.");
164
165 if (kinematicSystem.AxesLinear.Count <= 0 && kinematicSystem.AxesRotational.Count <= 0)
166 return new ArgumentException(
167 $"Kinematics must have at least 1 axis, either linear or rotational.");
168
169 bool canUseRequestedSpeedLinear = requestedSpeedLinear is not null && kinematicSystem.AxesLinear.Count > 0;
170 bool canUseRequestedSpeedRotational = requestedSpeedRotational is not null && kinematicSystem.AxesRotational.Count > 0;
171
172 if (!canUseRequestedSpeedLinear && !canUseRequestedSpeedRotational)
173 return new ArgumentException(
174 $"Kinematics must have at a linear axis and a requested linear speed or have a rotational axis and a rotational speed.");
175
176 // Ensure destination of all axes are within their travel limits and affected axes are homed.
177 for (int i = 0; i < kinematicSystem.AxesLinear.Count; i++)
178 {
179 Length? destination = finalAxisPositionsLinear[i];
180 AxisLinear axis = kinematicSystem.AxesLinear[i];
181 if (destination is not null && axis.PosOutOfRange(destination.Value))
182 return new ArgumentException($"Final position ({destination}) for linear axis '{axis.Name}' is out of bounds.");
183
184 if (!axis.IsHomed)
185 return new ArgumentException($"Linear axis '{axis.Name}' must be homed before it can be moved.");
186 }
187
188 for (int i = 0; i < kinematicSystem.AxesRotational.Count; i++)
189 {
190 Angle? destination = finalAxisPositionsRotational[i];
191 AxisRotational axis = kinematicSystem.AxesRotational[i];
192 if (destination is not null && axis.PosOutOfRange(destination.Value))
193 return new ArgumentException($"Final position ({destination}) for rotational axis '{axis.Name}' is out of bounds.");
194
195 if (!axis.IsHomed)
196 return new ArgumentException($"Rotational axis '{axis.Name}' must be homed before it can be moved.");
197 }
198
199 // get component distances
200 // distance = final - initial
201 List<Length?> axisComponentDistancesLinear = [.. finalAxisPositionsLinear.Zip(initialAxisPositionsLinear, (final, initial) => final - initial)];
202 List<Angle?> axisComponentDistancesRotational = [.. finalAxisPositionsRotational.Zip(initialAxisPositionsRotational, (final, initial) => final - initial)];
203
204
205 // get hypotenuse distances
206 // hypotenuse distance = Sqrt(d1^2 + d2^2 + ...)
207 Length axisHypotenuseDistanceLinear = UnitsNetHelpers.Helpers.Distance(axisComponentDistancesLinear);
208 Angle axisHypotenuseDistanceRotational = UnitsNetHelpers.Helpers.Distance(axisComponentDistancesRotational);
209
210 if (axisHypotenuseDistanceLinear.Equals(Length.Zero, precisions.PrecisionLinear) &&
211 axisHypotenuseDistanceRotational.Equals(Angle.Zero, precisions.PrecisionRotational))
212 return new ArgumentException(
213 $"Move must have a distance of at least {precisions.PrecisionLinear} or {precisions.PrecisionRotational}.");
214
215 int segmentCount = Max(
216 (int)Ceiling(axisHypotenuseDistanceLinear / moveSegmentLengthLinear),
217 (int)Ceiling(axisHypotenuseDistanceRotational / moveSegmentLengthRotational));
218
219 int pointCount = segmentCount + 1;
220
221 Debug.Assert(segmentCount > 0, "There must be at least one segment.");
222
223 //List<MoveSegment> segments = new(segmentCount);
224 List<MovePoint> points = new(pointCount);
225
226 for (int i = 0; i < pointCount; i++)
227 {
228 double pathRatio = (double)i / pointCount;
229
230 // get axis positions at each point
231 IImmutableList<Length?> currentAxisPositionsLinear = [.. axisComponentDistancesLinear
232 .Zip(initialAxisPositionsLinear, (distance, initialPos) => initialPos + pathRatio * distance)];
233
234 IImmutableList<Angle?> currentAxisPositionsRotational = [.. axisComponentDistancesRotational
235 .Zip(initialAxisPositionsRotational, (distance, initialPos) => initialPos + pathRatio * distance)];
236
237 // do inverse kinematics to get actuator positions at each point
238 Result<(IImmutableList<Length?>, IImmutableList<Angle?>)> inverseKinematics =
239 kinematicSystem.InverseKinematics(currentAxisPositionsLinear, currentAxisPositionsRotational);
240
241 if (inverseKinematics.IsError)
242 return inverseKinematics.Exception;
243
244 (IImmutableList<Length?> currentActuatorPositionsLinear,
245 IImmutableList<Angle?> currentActuatorPositionsRotational) = inverseKinematics.Value;
246
247
248 MovePoint point = new(
249 pathRatio,
250 currentAxisPositionsLinear,
251 currentAxisPositionsRotational,
252 currentActuatorPositionsLinear,
253 currentActuatorPositionsRotational);
254
255 points.Add(point);
256 }
257
258 return new Move(
259 precisions,
260 kinematicSystem,
261 requestedSpeedLinear,
262 requestedSpeedRotational,
263 axisComponentDistancesLinear,
264 axisComponentDistancesRotational,
265 axisHypotenuseDistanceLinear,
266 axisHypotenuseDistanceRotational,
267 points);
268 }

290 private static void CalculateActuatorDerivatives(IList<MovePoint> points)
291 {
292 int actuatorCountLinear = points[0].ActuatorPositionsLinear.Count;
293 int actuatorCountRotational = points[0].ActuatorPositionsRotational.Count;
294
295 for (int iPoint = 0; iPoint < points.Count; iPoint++)
296 {
297 MovePoint currentPoint = points[iPoint];
298 if (iPoint == 0)
299 {
300 // Derivatives of the first point are calculated with forward difference.
301 // dƟ/ds ≈ (ƟNext - ƟCurrent) / (sNext - sCurrent).
302 MovePoint nextPoint = points[iPoint + 1];
303 double ds = nextPoint.PathRatio - currentPoint.PathRatio;
304
305 // TODO: figure out what Claude means by this:
306 // Second derivative at the very start of the move mainly matters when
307 // path velocity is already nonzero there (e.g. a blended junction).
308 // At a standstill start the curvature term drops out of the acceleration
309 // constraint anyway (it is multiplied by velocity squared), so copying the
310 // nearest interior estimate is a safe, simple approximation.
311
312 for (int iActuator = 0; iActuator < actuatorCountLinear; iActuator++)
313 {
314 Length? currentPos = currentPoint.ActuatorPositionsLinear[iActuator];
315 Length? nextPos = nextPoint.ActuatorPositionsLinear[iActuator];
316 currentPoint.ActuatorPositionDerivatives1Linear[iActuator] = (nextPos - currentPos) / ds;
317 currentPoint.ActuatorPositionDerivatives2Linear[iActuator] = currentPos.HasValue ? Length.Zero : null;
318 }
319
320 for (int iActuator = 0; iActuator < actuatorCountRotational; iActuator++)
321 {
322 Angle? currentPos = currentPoint.ActuatorPositionsRotational[iActuator];
323 Angle? nextPos = nextPoint.ActuatorPositionsRotational[iActuator];
324 currentPoint.ActuatorPositionDerivatives1Rotational[iActuator] = (nextPos - currentPos) / ds;
325 currentPoint.ActuatorPositionDerivatives2Rotational[iActuator] = currentPos.HasValue ? Angle.Zero : null;
326 }
327 }
328 else if (iPoint == points.Count - 1)
329 {
330 // Derivatives of the last point are calculated with backward difference.
331 // dƟ/ds ≈ (ƟCurrent - ƟPrevious) / (sCurrent - sPrevious).
332 MovePoint previousPoint = points[iPoint - 1];
333 double ds = currentPoint.PathRatio - previousPoint.PathRatio;
334
335 for (int iActuator = 0; iActuator < actuatorCountLinear; iActuator++)
336 {
337 Length? currentPos = currentPoint.ActuatorPositionsLinear[iActuator];
338 Length? previousPos = previousPoint.ActuatorPositionsLinear[iActuator];
339 currentPoint.ActuatorPositionDerivatives1Linear[iActuator] = (currentPos - previousPos) / ds;
340 currentPoint.ActuatorPositionDerivatives2Linear[iActuator] = currentPos.HasValue ? Length.Zero : null;
341 }
342
343 for (int iActuator = 0; iActuator < actuatorCountRotational; iActuator++)
344 {
345 Angle? currentPos = currentPoint.ActuatorPositionsRotational[iActuator];
346 Angle? previousPos = previousPoint.ActuatorPositionsRotational[iActuator];
347 currentPoint.ActuatorPositionDerivatives1Rotational[iActuator] = (currentPos - previousPos) / ds;
348 currentPoint.ActuatorPositionDerivatives2Rotational[iActuator] = currentPos.HasValue ? Angle.Zero : null;
349 }
350 }
351 else
352 {
353 // Derivatives of all other points are calculated with central difference.
354 // The spacing between points is constant in axis space but not in actuator space so
355 // we will use the formula for unequal point spacing.
356 MovePoint previousPoint = points[iPoint - 1];
357 MovePoint nextPoint = points[iPoint + 1];
358 double dsBehind = currentPoint.PathRatio - previousPoint.PathRatio;
359 double dsAhead = nextPoint.PathRatio - currentPoint.PathRatio;
360
361 for (int iActuator = 0; iActuator < actuatorCountLinear; iActuator++)
362 {
363 Length? previousPos = previousPoint.ActuatorPositionsLinear[iActuator];
364 Length? currentPos = currentPoint.ActuatorPositionsLinear[iActuator];
365 Length? nextPos = nextPoint.ActuatorPositionsLinear[iActuator];
366 currentPoint.ActuatorPositionDerivatives1Linear[iActuator] =
367 (dsBehind * dsBehind * (nextPos - currentPos) + dsAhead * dsAhead * (currentPos - previousPos))
368 / (dsBehind * dsAhead * (dsBehind + dsAhead));
369 currentPoint.ActuatorPositionDerivatives2Linear[iActuator] =
370 2.0 * (dsBehind * (nextPos - currentPos) - dsAhead * (currentPos - previousPos))
371 / (dsBehind * dsAhead * (dsBehind + dsAhead));
372 }
373
374 for (int iActuator = 0; iActuator < actuatorCountRotational; iActuator++)
375 {
376 Angle? previousPos = previousPoint.ActuatorPositionsRotational[iActuator];
377 Angle? currentPos = currentPoint.ActuatorPositionsRotational[iActuator];
378 Angle? nextPos = nextPoint.ActuatorPositionsRotational[iActuator];
379 currentPoint.ActuatorPositionDerivatives1Rotational[iActuator] =
380 (dsBehind * dsBehind * (nextPos - currentPos) + dsAhead * dsAhead * (currentPos - previousPos))
381 / (dsBehind * dsAhead * (dsBehind + dsAhead));
382 currentPoint.ActuatorPositionDerivatives2Rotational[iActuator] =
383 2.0 * (dsBehind * (nextPos - currentPos) - dsAhead * (currentPos - previousPos))
384 / (dsBehind * dsAhead * (dsBehind + dsAhead));
385 }
386 }
387 }

396 private static void CalculateVelocityLimits(IList<MovePoint> points, CompositeKinematicSystem kinematicSystem)
397 {
398 int actuatorCountLinear = points[0].ActuatorPositionsLinear.Count;
399 int actuatorCountRotational = points[0].ActuatorPositionsRotational.Count;
400
401 Debug.Assert(kinematicSystem.ActuatorsLinear.Count == actuatorCountLinear);
402 Debug.Assert(kinematicSystem.ActuatorsRotational.Count == actuatorCountRotational);
403
404 for (int iPoint = 0; iPoint < points.Count; iPoint++)
405 {
406 MovePoint currentPoint = points[iPoint];
407 Frequency actuatorLimitedDsDt = Frequency.FromHertz(double.MaxValue);
408
409 for (int iActuator = 0; iActuator < actuatorCountLinear; iActuator++)
410 {
411 Length? firstDerivative = currentPoint.ActuatorPositionDerivatives1Linear[iActuator];
412
413 if (!firstDerivative.HasValue)
414 continue;
415
416 // The built-in UnitsNet Abs() function is super slow for some reason...
417 Length firstDerivativeMagnitude = Length.FromMillimeters(Abs(firstDerivative.Value.Millimeters));
418
419 if (firstDerivativeMagnitude > Constants.VERY_SMALL_LENGTH)
420 {
421 IActuatorLinear actuator = kinematicSystem.ActuatorsLinear[iActuator];
422 Frequency candidateDsDt = actuator.MaxSpeed / firstDerivativeMagnitude;
423 actuatorLimitedDsDt = UnitsNetHelpers.Helpers.Min(actuatorLimitedDsDt, candidateDsDt);
424 }
425 }
426
427 for (int iActuator = 0; iActuator < actuatorCountRotational; iActuator++)
428 {
429 Angle? firstDerivative = currentPoint.ActuatorPositionDerivatives1Rotational[iActuator];
430
431 if (!firstDerivative.HasValue)
432 continue;
433
434 // The built-in UnitsNet Abs() function is super slow for some reason...
435 Angle firstDerivativeMagnitude = Angle.FromRadians(Abs(firstDerivative.Value.Radians));
436
437 if (firstDerivativeMagnitude > Constants.VERY_SMALL_ANGLE)
438 {
439 IActuatorRotational actuator = kinematicSystem.ActuatorsRotational[iActuator];
440 Frequency candidateDsDt = actuator.MaxSpeed / firstDerivativeMagnitude;
441 actuatorLimitedDsDt = UnitsNetHelpers.Helpers.Min(actuatorLimitedDsDt, candidateDsDt);
442 }
443 }
444
445 currentPoint.ActuatorLimitedDsDt = actuatorLimitedDsDt;
446 }

449 private static void CalculateDsDtValues(IList<MovePoint> points, CompositeKinematicSystem kinematicSystem)
450 {
451 // The time-optimal profile at every sample is the lower envelope of the two passes:
452 // It must be slow enough that the forward pass could have reached it, and slow
453 // enough that the backward pass could still decelerate to the required end condition.
454 Frequency[] forwardPassDsDt = new Frequency[points.Count];
455 Frequency[] backwardPassDsDt = new Frequency[points.Count];
456
457 //TODO: make this work with non-zero start and end velocities. In other words allow junction velocities to be used.
458 forwardPassDsDt[0] = Frequency.Zero;
459 for (int i = 0; i < points.Count - 1; i++)
460 {
461 MovePoint currentPoint = points[i];
462 MovePoint nextPoint = points[i + 1];
463 double ds = nextPoint.PathRatio - currentPoint.PathRatio;
464 Frequency2 maxPathD2sDt2 = CalculateMaxPathD2sDt2(
465 currentPoint, forwardPassDsDt[i], kinematicSystem, returnMinPathAcceleration: false);
466 Frequency2 candidateNextD2sDt2 = forwardPassDsDt[i] * forwardPassDsDt[i] + 2.0 * maxPathD2sDt2 * ds;
467 Frequency candidateNextDsDt = candidateNextD2sDt2 > Frequency2.Zero ? candidateNextD2sDt2.Sqrt() : Frequency.Zero;
468
469 forwardPassDsDt[i + 1] = UnitsNetHelpers.Helpers.Min(candidateNextDsDt, nextPoint.ActuatorLimitedDsDt);
470 }
471
472 backwardPassDsDt[^1] = Frequency.Zero;
473 for (int i = points.Count - 1; i > 0; i--)
474 {
475 MovePoint previousPoint = points[i - 1];
476 MovePoint currentPoint = points[i];
477
478 double ds = currentPoint.PathRatio - previousPoint.PathRatio;
479 Frequency2 minPathD2sDt2 = CalculateMaxPathD2sDt2(
480 currentPoint, backwardPassDsDt[i], kinematicSystem, returnMinPathAcceleration: true);
481 Frequency2 minPathD2sDt2Magnitude = minPathD2sDt2.Abs();
482 Frequency2 candidatePreviousD2sDt2 = backwardPassDsDt[i] * backwardPassDsDt[i] + 2.0 * minPathD2sDt2Magnitude * ds;
483 Frequency candidatePreviousDsDt = candidatePreviousD2sDt2 > Frequency2.Zero ? candidatePreviousD2sDt2.Sqrt() : Frequency.Zero;
484
485 backwardPassDsDt[i - 1] = UnitsNetHelpers.Helpers.Min(candidatePreviousDsDt, previousPoint.ActuatorLimitedDsDt);
486 }
487
488 for (int i = 0; i < points.Count - 1; i++)
489 points[i].ActualDsDt = UnitsNetHelpers.Helpers.Min(forwardPassDsDt[i], backwardPassDsDt[i]);
490 }

492 private static Frequency2 CalculateMaxPathD2sDt2(
493 MovePoint point,
494 Frequency currentDsDt,
495 CompositeKinematicSystem kinematicSystem,
496 bool returnMinPathAcceleration)
497 {
498 Frequency2 minimumPathAcceleration = Frequency2.FromHertzSquared(double.MinValue);
499 Frequency2 maximumPathAcceleration = Frequency2.FromHertzSquared(double.MaxValue);
500
501 int actuatorCountLinear = point.ActuatorPositionsLinear.Count;
502 int actuatorCountRotational = point.ActuatorPositionsRotational.Count;
503
504 Debug.Assert(kinematicSystem.ActuatorsLinear.Count == actuatorCountLinear);
505 Debug.Assert(kinematicSystem.ActuatorsRotational.Count == actuatorCountRotational);
506
507 for (int iActuator = 0; iActuator < actuatorCountLinear; iActuator++)
508 {
509 IActuatorLinear actuator = kinematicSystem.ActuatorsLinear[iActuator];
510 Length? firstDerivativeNullable = point.ActuatorPositionDerivatives1Linear[iActuator];
511 Length? secondDerivativeNullable = point.ActuatorPositionDerivatives2Linear[iActuator];
512
513 if (firstDerivativeNullable is null || secondDerivativeNullable is null)
514 continue;
515
516 Length firstDerivative = firstDerivativeNullable.Value;
517 Length secondDerivative = secondDerivativeNullable.Value;
518
519 // Path acceleration has (almost) no effect on this actuator's acceleration here so
520 // it does not constrain "a" at this sample.
521 if (firstDerivative.Equals(Length.Zero, Length.FromMillimeters(1e-9)))
522 continue;
523
524 Acceleration curvatureContribution = secondDerivative * currentDsDt * currentDsDt;
525
526 Acceleration lowerBoundOnActuatorAcceleration = -actuator.MaxAcceleration - curvatureContribution;
527 Acceleration upperBoundOnActuatorAcceleration = actuator.MaxAcceleration - curvatureContribution;
528
529 Frequency2 candidateBoundA = lowerBoundOnActuatorAcceleration / firstDerivative;
530 Frequency2 candidateBoundB = upperBoundOnActuatorAcceleration / firstDerivative;
531
532 Frequency2 actuatorMinimumPathAcceleration = UnitsNetHelpers.Helpers.Min(candidateBoundA, candidateBoundB);
533 Frequency2 actuatorMaximumPathAcceleration = UnitsNetHelpers.Helpers.Max(candidateBoundA, candidateBoundB);
534
535 minimumPathAcceleration = UnitsNetHelpers.Helpers.Max(minimumPathAcceleration, actuatorMinimumPathAcceleration);
536 maximumPathAcceleration = UnitsNetHelpers.Helpers.Min(maximumPathAcceleration, actuatorMaximumPathAcceleration);
537 }
538
539 for (int iActuator = 0; iActuator < actuatorCountRotational; iActuator++)
540 {
541 IActuatorRotational actuator = kinematicSystem.ActuatorsRotational[iActuator];
542 Angle? firstDerivativeNullable = point.ActuatorPositionDerivatives1Rotational[iActuator];
543 Angle? secondDerivativeNullable = point.ActuatorPositionDerivatives2Rotational[iActuator];
544
545 if (firstDerivativeNullable is null || secondDerivativeNullable is null)
546 continue;
547
548 Angle firstDerivative = firstDerivativeNullable.Value;
549 Angle secondDerivative = secondDerivativeNullable.Value;
550
551 // Path acceleration has (almost) no effect on this actuator's acceleration here so
552 // it does not constrain "a" at this sample.
553 if (firstDerivative.Equals(Angle.Zero, Angle.FromRadians(1e-9)))
554 continue;
555
556 RotationalAcceleration curvatureContribution = secondDerivative * currentDsDt * currentDsDt;
557
558 RotationalAcceleration lowerBoundOnActuatorAcceleration = -actuator.MaxAcceleration - curvatureContribution;
559 RotationalAcceleration upperBoundOnActuatorAcceleration = actuator.MaxAcceleration - curvatureContribution;
560
561 Frequency2 candidateBoundA = lowerBoundOnActuatorAcceleration / firstDerivative;
562 Frequency2 candidateBoundB = upperBoundOnActuatorAcceleration / firstDerivative;
563
564 Frequency2 actuatorMinimumPathAcceleration = UnitsNetHelpers.Helpers.Min(candidateBoundA, candidateBoundB);
565 Frequency2 actuatorMaximumPathAcceleration = UnitsNetHelpers.Helpers.Max(candidateBoundA, candidateBoundB);
566
567 minimumPathAcceleration = UnitsNetHelpers.Helpers.Max(minimumPathAcceleration, actuatorMinimumPathAcceleration);
568 maximumPathAcceleration = UnitsNetHelpers.Helpers.Min(maximumPathAcceleration, actuatorMaximumPathAcceleration);
569 }
570
571 return returnMinPathAcceleration ? minimumPathAcceleration : maximumPathAcceleration;
572 }

574 private static void CalculateTimes(IList<MovePoint> points)
575 {
576 points[0].Time = Duration.Zero;
577
578 for (int i = 0; i < points.Count - 1; i++)
579 {
580 MovePoint currentPoint = points[i];
581 MovePoint nextPoint = points[i + 1];
582 double ds = nextPoint.PathRatio - currentPoint.PathRatio;
583 Frequency averageDsDtForThisInterval = (currentPoint.ActualDsDt + nextPoint.ActualDsDt) / 2.0;
584 Duration timeForThisInterval = averageDsDtForThisInterval > Constants.VERY_SMALL_FREQUENCY
585 ? ds / averageDsDtForThisInterval
586 : Duration.Zero;
587 nextPoint.Time = currentPoint.Time + timeForThisInterval;
588 }

631 private static void SimplifyMoveSegment(
632 HashSet<int> pointIndiciesToKeep,
633 Move move,
634 int startIndex,
635 int endIndex,
636 Length actuatorEpsilonLinear,
637 Angle actuatorEpsilonRotational)
638 {
639 // Endpoints will always be included in simplification.
640 pointIndiciesToKeep.Add(startIndex);
641 pointIndiciesToKeep.Add(endIndex);
642
643 int numberOfPoints = endIndex - startIndex + 1;
644
645 // Not enough points to simplify further.
646 if (numberOfPoints <= 2)
647 return;
648
649 MovePoint startPoint = move.Points[startIndex];
650 MovePoint endPoint = move.Points[endIndex];
651 Duration timeRange = endPoint.Time - startPoint.Time;
652 IReadOnlyList<Length?> actuatorDistancesLinear = [.. endPoint.ActuatorPositionsLinear
653 .Zip(startPoint.ActuatorPositionsLinear, (end, start) => end - start)];
654 IReadOnlyList<Angle?> actuatorDistancesRotational = [.. endPoint.ActuatorPositionsRotational
655 .Zip(startPoint.ActuatorPositionsRotational, (end, start) => end - start)];
656
657
658 // This keeps track of how far off the current point is relative to the provided epsilons.
659 double maxEpsilonRatio = 0.0;
660 int indexWithMaxEpsilonRatio = 0;
661
662 // Loop through each point and find the point where any actuator deviates the most.
663 for (int i = startIndex; i <= endIndex; i++)
664 {
665 MovePoint currentPoint = move.Points[i];
666 double timeRatio = (currentPoint.Time - startPoint.Time) / timeRange;
667 for (int iActuator = 0; iActuator < move.KinematicSystem.ActuatorsLinear.Count; iActuator++)
668 {
669 Length? trueActuatorPosition = currentPoint.ActuatorPositionsLinear[iActuator];
670 Length? approximateActuatorPosition = startPoint.ActuatorPositionsLinear[iActuator] +
671 timeRatio * actuatorDistancesLinear[iActuator];

673

682 }
683 }
684 }

688 {
689 MovePoint currentPoint = move.Points[i];
690 double timeRatio = (currentPoint.Time - startPoint.Time) / timeRange;
691 for (int iActuator = 0; iActuator < move.KinematicSystem.ActuatorsRotational.Count; iActuator++)
692 {
693 Angle? trueActuatorPosition = currentPoint.ActuatorPositionsRotational[iActuator];
694 Angle? approximateActuatorPosition = startPoint.ActuatorPositionsRotational[iActuator] +
695 timeRatio * actuatorDistancesRotational[iActuator];
696 Angle? epsilon = approximateActuatorPosition - trueActuatorPosition;
697
698 if (!epsilon.HasValue)
699 continue;
700
701 double epsilonRatio = epsilon.Value.FastAbs() / actuatorEpsilonRotational;
702 if (epsilonRatio > maxEpsilonRatio)
703 {
704 maxEpsilonRatio = epsilonRatio;
705 indexWithMaxEpsilonRatio = i;
706 }
707 }
708 }
709
710 if (maxEpsilonRatio > 1.0)
711 {
712 // Straight line approximation is not enough so keep the point with the highest error then
713 // split into two segments at that index and run again recursively.
714 SimplifyMoveSegment(
715 pointIndiciesToKeep,
716 move,
717 startIndex,
718 indexWithMaxEpsilonRatio,
719 actuatorEpsilonLinear,

728 }
