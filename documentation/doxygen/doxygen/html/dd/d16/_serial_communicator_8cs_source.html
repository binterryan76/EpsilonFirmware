139 public override IEnumerable<DataPacket> Read()
140 {
141 List<DataPacket> packets = [];
142
143 if (!connected)
144 {
145 bool success = Connect();
146 if (!success)
147 return packets;
148 }
149
150 try
151 {
152 while (serialPort.BytesToRead > 0)
153 {
154 // ensure at least one byte is available to read
155 int bytesThatCanBeRead = serialPort.BytesToRead;
156
157 if (!dataPacketStarted)
158 {
159 // first byte is the data length
160 int packetDataLength = serialPort.ReadByte();
161 bytesThatCanBeRead--;
162
163 if (packetDataLength > DataPacket.MAX_DATA_LENGTH)
164 throw new Exception($"Data length cannot exceed {DataPacket.MAX_DATA_LENGTH}.");
165
166 if (packetDataLength < DataPacket.MIN_DATA_LENGTH)
167 throw new Exception($"Data length must be at least {DataPacket.MIN_DATA_LENGTH}.");
168
169 if (bytesThatCanBeRead >= packetDataLength)
170 {
171 // yay we can read entire packet and return it
172 uint totalLength = (uint)(packetDataLength + 1);
173 byte[] allData = new byte[totalLength];
174
175 // set data length byte
176 allData[0] = (byte)packetDataLength;
177
178 // set remaining bytes
179 serialPort.Read(allData, 1, packetDataLength);
180
181 // initialize new packet (this sets first byte of array)
182 Result<DataPacket> dataPacket = DataPacket.New(Id, allData);
183
184 Debug.Assert(dataPacket.IsSuccess);
185
186 dataPacketStarted = false;
187 Debug.WriteLine(dataPacket.ToString());
188 packets.Add(dataPacket.Value);
189 continue;
190 }
191 else
192 {
193 // initialize packet (this sets first byte of array)
194 startedDataPacketBytes = new byte[packetDataLength + 1];
195 startedDataPacketBytes[0] = (byte)packetDataLength;
196
197 // read as many remaining bytes as we can
198 serialPort.Read(startedDataPacketBytes, 1, bytesThatCanBeRead);
199
200 remainingBytesToRead = packetDataLength - bytesThatCanBeRead;
201 dataPacketStarted = true;
202 return packets;
203 }
204 }
205 else // data packet has been started
206 {
207 // figure out offset of where we will write new bytes
208 byte numDataBytes = startedDataPacketBytes[0];
209 int offset = numDataBytes + 1 - remainingBytesToRead;
210
211 if (bytesThatCanBeRead >= remainingBytesToRead)
212 {
213 // yay we can read rest of packet and return it
214 // set remaining bytes
215 serialPort.Read(startedDataPacketBytes, offset, remainingBytesToRead);
216
217 dataPacketStarted = false;
218
219 Result<DataPacket> dataPacket = DataPacket.New(Id, startedDataPacketBytes);
220
221 Debug.Assert(dataPacket.IsSuccess);
222
223 Debug.WriteLine(dataPacket.Value.ToString());
224 packets.Add(dataPacket.Value);
225 continue;
226 }
227 else
228 {
229 // just add as many more bytes as we can
230 serialPort.Read(startedDataPacketBytes, offset, bytesThatCanBeRead);
231
232 remainingBytesToRead -= bytesThatCanBeRead;
233 dataPacketStarted = true;
234 return packets;
235 }
236 }
237 }
238 return packets;
239 }
240 catch (TimeoutException)
241 {
242 Debug.WriteLine($"Read timeout - {serialPort.PortName}");
243 return packets;
244 }
245 catch (Exception ex)
246 {
247 Debug.WriteLine($"Error reading from serial port - {serialPort.PortName}: {ex.Message}");
248 return packets;
249 }
